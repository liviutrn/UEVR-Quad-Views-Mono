if(WIN32)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/DuoViewShaderSource.hpp"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/embed_shader.py" "${CMAKE_CURRENT_BINARY_DIR}/DuoViewShaderSource.hpp" --include-dibr
    DEPENDS embed_shader.py DuoViewComposite.hlsl DuoViewCompositeRaster.hlsl DuoViewNativeCompositeRaster.hlsl DuoViewGlamaryeFovea.hlsl dibr/DuoViewDibrRaster.hlsl)
  add_executable(DuoViewProductionWarpTests production_warp_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/DuoViewShaderSource.hpp")
  target_include_directories(DuoViewProductionWarpTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(DuoViewProductionWarpTests PRIVATE d3d12 dxgi d3dcompiler)
  add_executable(DuoViewManualDibrWarpTests dibr/manual_dibr_warp_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/DuoViewShaderSource.hpp")
  target_include_directories(DuoViewManualDibrWarpTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(DuoViewManualDibrWarpTests PRIVATE d3d12 dxgi d3dcompiler)
  add_executable(DuoViewManualDibrCaptureWarpTests dibr/manual_dibr_capture_warp_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/DuoViewShaderSource.hpp")
  target_include_directories(DuoViewManualDibrCaptureWarpTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(DuoViewManualDibrCaptureWarpTests PRIVATE d3d12 dxgi)
  if(MSVC)
    target_compile_options(DuoViewManualDibrCaptureWarpTests PRIVATE /UNDEBUG)
  else()
    target_compile_options(DuoViewManualDibrCaptureWarpTests PRIVATE -UNDEBUG)
  endif()
  add_executable(DuoViewWarpTests DuoViewWarpTest.cpp)
  target_link_libraries(DuoViewWarpTests PRIVATE d3d12 dxgi d3dcompiler)
endif()

  if(MSVC)
    target_compile_options(${test} PRIVATE /UNDEBUG)
  else()

  endif()
