cmake_minimum_required(VERSION 3.20)
project(DuoViewAfwAbiProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "AFWAbiProbe must be built on 64-bit Windows")
endif()
add_executable(AFWAbiProbe afw_abi_probe.cpp)
target_compile_features(AFWAbiProbe PRIVATE cxx_std_17)
target_compile_definitions(AFWAbiProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(AFWAbiProbe PRIVATE /W4 /WX /utf-8)
endif()
enable_testing()
add_test(NAME afw_abi_selftest COMMAND $<TARGET_FILE:AFWAbiProbe> --self-test)
