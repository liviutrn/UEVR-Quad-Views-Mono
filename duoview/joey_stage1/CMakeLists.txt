cmake_minimum_required(VERSION 3.20)
project(DuoViewJoeyPinnedContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT JOYEY_BACKEND_DIR)
  message(FATAL_ERROR "Pass JOYEY_BACKEND_DIR with exact patched Joey backend")
endif()
add_executable(DuoViewPinnedProductionContracts ProductionContractTests.cpp)
target_include_directories(DuoViewPinnedProductionContracts PRIVATE "${JOYEY_BACKEND_DIR}/src/mods/vr")
enable_testing()
add_test(NAME pinned-duoview-frame-and-layout COMMAND DuoViewPinnedProductionContracts)

if(WIN32)
  add_executable(DuoViewPinnedPSVR2Gaze ProductionPSVR2GazeTests.cpp)
  target_include_directories(DuoViewPinnedPSVR2Gaze PRIVATE "${JOYEY_BACKEND_DIR}/src/mods/vr")
  target_link_libraries(DuoViewPinnedPSVR2Gaze PRIVATE Advapi32)
  add_test(NAME pinned-psvr2-capi-projection-and-validation COMMAND DuoViewPinnedPSVR2Gaze)
  add_executable(DuoViewPinnedProductionWarp ProductionWarpPinned.cpp)
  target_include_directories(DuoViewPinnedProductionWarp PRIVATE "${JOYEY_BACKEND_DIR}/src/mods/vr")
  target_link_libraries(DuoViewPinnedProductionWarp PRIVATE d3d12 dxgi d3dcompiler)
  add_test(NAME pinned-duoview-dx12-warp COMMAND DuoViewPinnedProductionWarp)
  set_tests_properties(pinned-duoview-dx12-warp PROPERTIES TIMEOUT 60)
  # v208: shared compose texture + "same image for both eyes" pixel contract.
  add_executable(DuoViewV208SharedEyeWarp V208SharedEyeWarp.cpp)
  target_include_directories(DuoViewV208SharedEyeWarp PRIVATE "${JOYEY_BACKEND_DIR}/src/mods/vr")
  target_link_libraries(DuoViewV208SharedEyeWarp PRIVATE d3d12 dxgi d3dcompiler)
  add_test(NAME v208-shared-eye-dx12-warp COMMAND DuoViewV208SharedEyeWarp)
  set_tests_properties(v208-shared-eye-dx12-warp PROPERTIES TIMEOUT 60)
endif()

# v208: shared OpenXR rectangle covers both eye crops with its exact FOV.
add_executable(DuoViewV208SharedCrop V208SharedCropTests.cpp)
target_include_directories(DuoViewV208SharedCrop PRIVATE "${JOYEY_BACKEND_DIR}/src/mods/vr")
add_test(NAME v208-shared-eye-crop COMMAND DuoViewV208SharedCrop)

find_package(Threads REQUIRED)
add_executable(DuoViewDeliveryTests DeliveryTests.cpp)
target_link_libraries(DuoViewDeliveryTests PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(DuoViewDeliveryTests PRIVATE /UNDEBUG)
else()
  target_compile_options(DuoViewDeliveryTests PRIVATE -UNDEBUG)
endif()
add_test(NAME pinned-duoview-asynchronous-delivery COMMAND DuoViewDeliveryTests)
