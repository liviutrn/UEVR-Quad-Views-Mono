cmake_minimum_required(VERSION 3.20)
project(TriViewStage1Tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# TRIVIEW_INCLUDE_DIR = patched backend src/mods/vr (holds the generated TriViewShaderSource.hpp).
if(NOT TRIVIEW_INCLUDE_DIR)
  message(FATAL_ERROR "Pass TRIVIEW_INCLUDE_DIR (patched backend src/mods/vr)")
endif()
enable_testing()
add_executable(TriViewGeometryTests TriViewGeometryTests.cpp)
target_include_directories(TriViewGeometryTests PRIVATE "${TRIVIEW_INCLUDE_DIR}")
add_test(NAME triview-geometry COMMAND TriViewGeometryTests)
add_executable(TriViewCaptureTests TriViewCaptureTests.cpp)
add_test(NAME triview-capture-writer COMMAND TriViewCaptureTests)
if(WIN32)
  add_executable(TriViewCompositorWarpTests TriViewCompositorWarpTests.cpp)
  target_include_directories(TriViewCompositorWarpTests PRIVATE "${TRIVIEW_INCLUDE_DIR}")
  target_link_libraries(TriViewCompositorWarpTests PRIVATE d3d12 dxgi)
  add_test(NAME triview-compositor-warp COMMAND TriViewCompositorWarpTests)
  set_tests_properties(triview-compositor-warp PROPERTIES TIMEOUT 60)
endif()
